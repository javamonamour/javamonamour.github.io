connect("weblogic", "weblogic1", "t3://hostname.acme.com:7001")
cd('/')
SERVERNAME='osbts2ms2'
HOST='hostname.acme.com'
PORTNUMBER='8007'
edit()
startEdit()
create(SERVERNAME, 'Server')
cd('Servers/'+SERVERNAME)
set('ListenAddress',HOST)
set('ListenPort',PORTNUMBER)
set('Machine', HOST)
save()
validate()
activate()
