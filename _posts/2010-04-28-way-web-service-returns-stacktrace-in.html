SOAPFault f = e.getFault();
Detail newDetail = f.getDetail();
if (newDetail != null) {
Iterator entries = newDetail.getDetailEntries();
while ( entries.hasNext() ) {
DetailEntry newEntry = (DetailEntry)entries.next();
String value = newEntry.getValue();
e.getCause().initCause(new IllegalArgumentException(value));
}
Throwable remoteException = e.getCause();
e.printStackTrace();
