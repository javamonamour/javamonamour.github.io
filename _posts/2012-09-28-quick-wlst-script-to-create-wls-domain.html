--- layout: post title: "Quick WLST script to create WLS domain using a template" date: 2012-09-28T03:09:00.001+02:00 permalink: /2012/09/quick-wlst-script-to-create-wls-domain.html blogger_id: "976657044900203816" tags: ["WLST"] --- readTemplate('/opt/oracle/fmw11_1_1_5/wlserver_10.3/common/templates/domains/wls.jar')
cd('Servers/AdminServer')
set('ListenAddress',''myhostname.com'')
set('ListenPort', 7001)
cd('/')
cd('Security/base_domain/User/weblogic')
cmo.setPassword('weblogic1')
setOption('OverwriteDomain', 'true')
writeDomain('/opt/oracle/fmw11_1_1_5/user_projects/domains/mydomain')
closeTemplate()
exit()
