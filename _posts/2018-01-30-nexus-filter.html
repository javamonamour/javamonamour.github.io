docker pull sonatype/nexus

docker run -d -p 8081:8081 --name nexus2 sonatype/nexus:oss

http://localhost:8081/nexus/ login as admin/admin123

curl http://localhost:8081/nexus/service/local/status


in your maven /home/centos/apache-maven-3.5.0/conf/settings.xml place this for Nexus 3:

<mirror>
<id>nexus</id>
<mirrorOf>*</mirrorOf>
<url>http://localhost:8081/repository/maven-public/</url>
</mirror>

or this for Nexus 2:

<mirror>
<id>nexus</id>
<mirrorOf>*</mirrorOf>
<url>http://localhost:8081/nexus/content/repositories/central/</url>
</mirror>
