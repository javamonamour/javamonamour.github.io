--- layout: post title: "Bash script to find the location of the currently running script" date: 2010-02-10T17:13:00.001+01:00 permalink: /2010/02/bash-script-to-find-location-of.html blogger_id: "1761314704041948095" tags: ["bash", "dirname", "readlink"] --- #!/bin/bash
#==========================
# bash - find path to script
#==========================
abspath="$(cd "${0%/*}" 2>/dev/null; echo "$PWD"/"${0##*/}")"

# to get the path only - not the script name - add
path_only=`dirname "$abspath"`

#display the paths to prove it works
echo $abspath
echo $path_only

SCRIPT=`readlink -f $0`
