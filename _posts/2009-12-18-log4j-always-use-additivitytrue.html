--- layout: post title: "log4j: always use additivity=false" date: 2009-12-18T14:32:00.000+01:00 permalink: /2009/12/log4j-always-use-additivitytrue.html blogger_id: "266975713887920319" --- the logging code synchronizes on each category; you waste a lot of time - and risk to create serious contention - if you let every category to bubble up to "root".


Category.callAppenders :

  public void callAppenders(LoggingEvent event)   {
    int writes = 0;

    for (Category c = this; c != null; c = c.parent)    {
      synchronized (c) {
        if (c.aai != null)
          writes += c.aai.appendLoopOnAppenders(event);

        if (!(c.additive)) {
          monitorexit; break label67:
        }
      }
    }

    if (writes == 0)
      label67: this.repository.emitNoAppenderWarning(this);
  }
